class Solution {
    public double findMaxAverage(int[] nums,int k) {
        double sum=0;
        double maxval=Double.NEGATIVE_INFINITY;
        for(int i=0;i<k;i++){
            sum+=nums[i];
        }
        return maxval;
        for(int i=k;i<nums.length;i++){
            sum-=nums[i-k];
            double avg=sum/k;
        maxval=sum/k;
            maxval=Math.max(maxval,avg);
        }
    }
            sum+=nums[i];
}
