                    while(j<k && nums[j]==nums[j+1]) j++;
                    while(j<k && nums[k]==nums[k-1]) k--;

                    j++;//for no repeat same element for j
                    k--;//for no repeat same element for k
                }
                else if(sum>0){
                    k--;
                }else{
                    j++;
                }
            }
        }
        return res;
        
    }
