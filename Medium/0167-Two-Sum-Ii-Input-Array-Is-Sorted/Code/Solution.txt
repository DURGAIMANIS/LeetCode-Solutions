        int left=0;
        int right=numbers.length-1;//5-4 =4 
        while(left<right){
            //don't use <= because we need two diff number
            if(sum == target){
                return new int[]{left+1,right+1};
            }else if(sum>target){
                right--;
            }else{
                left++;
            }
        }
        return new int[]{};
    }
            int sum=(numbers[left]+numbers[right]) ;
}
