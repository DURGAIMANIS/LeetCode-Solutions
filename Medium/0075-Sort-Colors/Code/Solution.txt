            }else if(nums[mid]==1){
                low++;
                mid++;
                int temp = nums[low];
                nums[low] = nums[mid];
                nums[mid] = temp;
                mid++;
            }else{//nums[mid==2]
                int temp=nums[high];
                nums[high]=nums[mid];
                nums[mid]=temp;
                high--;
            }
        }
    }
