class Solution {
    public int lengthOfLongestSubstring(String s) {
        HashSet<Character> seen=new HashSet<>();

        int start=0;
        int end=0;
        int maxlen=0;

        while(end<s.length()){
            char c=s.charAt(end);

            while(seen.contains(c)){
                seen.remove(s.charAt(start));
                start+=1;
            }
            seen.add(c);
