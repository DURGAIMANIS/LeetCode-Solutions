    public int minOperations(int[] nums) {
        int count=0;
        int k=3;
        for(int i=0;i<nums.length;i++){
            if(nums[i]==0){
                for(int j=i;j<k+i;j++){
            }
        }
                    if(nums[j]==0) nums[j]=1;
                }
                    else nums[j]=0;
                count++;
                if(i+2>=nums.length){
                    return -1;
                }
        return count;
    }
