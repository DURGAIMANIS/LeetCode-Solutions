
            if (right - left + 1 == k) {
                maxSum = Math.max(maxSum, sum);

                set.remove(nums[left]);
                sum -= nums[left];
                left++;
            sum += nums[right];
            set.add(nums[right]);

            }
                sum -= nums[left];
                left++;
                set.remove(nums[left]);
            }
        }

        return maxSum;
