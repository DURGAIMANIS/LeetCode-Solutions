            while(map.size()>2){
                map.put(fruits[left],map.get(fruits[left])-1);
                if(map.get(fruits[left])==0){
                    map.remove(fruits[left]);
                }
                left++;
            }
            map.put(fruits[i],map.getOrDefault(fruits[i],0)+1);
            ans=Math.max(ans,i-left+1);
        }
        return ans;
    }
}
