                currmax++;
            }else{
                currmax=lastcount+1;
            }

            if(fruit==lastfruit){
                lastcount++;
            }else{
                lastcount=1;
                secondlastfruit=lastfruit;
                lastfruit=fruit;
            }
            max=Math.max(max,currmax);
        }
        return max;
    }
}
