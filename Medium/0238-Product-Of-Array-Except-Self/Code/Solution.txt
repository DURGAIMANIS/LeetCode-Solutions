        int before = 1;

        for (int i = 0; i < nums.length; i++) {
            arr[i] = before;
            before *= nums[i];
        }

        // Product of all elements to the right
        int after = 1;

        for (int i = nums.length - 1; i >= 0; i--) {
            arr[i] *= after;
            after *= nums[i];
        }

        return arr;
