                water+=leftmax - height[left+1];
                left++;
        while(left<right){
            if(leftmax<rightmax){
                leftmax=Math.max(leftmax,height[left+1]);//(current max,nextvalue)

            }else{
                rightmax=Math.max(rightmax,height[right-1]);
                water+=rightmax- height[right-1];
                right--;
            }
        }
        return water;
    }
}
