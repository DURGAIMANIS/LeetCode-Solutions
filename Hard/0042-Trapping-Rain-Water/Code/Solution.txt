        while(left<right){
            if(leftmax<rightmax){

                leftmax=Math.max(leftmax,height[left+1]);
                water+=leftmax - height[left+1];
            }else{
        int water=0;

                rightmax=Math.max(rightmax,height[right-1]);
                water+=rightmax- height[right-1];
            }
        }
        return water;
                left++;
                right--;
    }
