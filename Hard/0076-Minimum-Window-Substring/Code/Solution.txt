                }
                char leftChar = s.charAt(left);
                window[leftChar]--;

                if (need[leftChar] > 0 && window[leftChar] < need[leftChar]) {//< only
                   required++;
                }
                left++;
            }
            right++;
        }
        if (minLength == Integer.MAX_VALUE) {
           return "";
        }
        return s.substring(start, start + minLength);
