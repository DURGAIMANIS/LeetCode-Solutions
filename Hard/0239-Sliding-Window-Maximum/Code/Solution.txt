                deque.pollFirst();
            }
            while(!deque.isEmpty()&&nums[deque.peekLast()]<nums[right]){
                deque.pollLast();
            while(!deque.isEmpty()&&deque.peekFirst()<=right-k){//remove from the first because of 
            maintan the window 
            }

            deque.addLast(right);//stores the index
            if(right>=k-1){
                result[right-k+1]=nums[deque.peekFirst()];
            }
        }
        return result;
    }
}
